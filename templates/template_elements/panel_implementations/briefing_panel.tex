% ----------------------------------------- Briefing Item ----------------------------------------- %
\NewDocumentCommand{\BriefingItem}{ O{} +m }{%
	% Always begin the item on a fresh line
	\par
	% ------------------------------------------------------------------------
	% Separator between entries
	% ------------------------------------------------------------------------
	\ifbool{bBriefingPanelFirstItem}{}{%
		\vspace{\BriefingPanelItemSep}
		{
			\color{briefingpanelblue!25}
			\hrule height 0.35pt
		}
		\vspace{\BriefingPanelItemSep}
	}%
	
	\boolfalse{bBriefingPanelFirstItem}
	% ------------------------------------------------------------------------
	% Entry
	% ------------------------------------------------------------------------
	\noindent
	\begin{minipage}[t]{\BriefingPanelSymbolWidth}

		\vspace{0pt}
		\centering
		\begin{tikzpicture}[x=1pt, y=1pt, baseline=-.5ex]
			\def\temp{#1}
			\ifthenelse{\equal{\temp}{}}%
			{
				\BriefingPanelItemSymbol%
			}
			{
				#1
			}
		\end{tikzpicture}

	\end{minipage}%
	\hspace{\BriefingPanelSymbolGap}%
	\begin{minipage}[t]
	{
		\dimexpr\linewidth
		- \BriefingPanelSymbolWidth
		- \BriefingPanelSymbolGap
		- \BriefingPanelItemRightPadding
		\relax
	}
		\vspace{0pt}
		#2
	\end{minipage}
	\par
}%

\ExplSyntaxOn
\NewDocumentEnvironment{BriefingPanel}{ O{} +b }{%
	\group_begin:
	% ------------------------------------------------------------------------
	% Defaults + Keys
	% ------------------------------------------------------------------------
	\BriefingPanelReset
	\keys_set:nn{ dnd / briefingpanel }{ #1 }
	% ------------------------------------------------------------------------
	% Calculate internal width
	% ------------------------------------------------------------------------
	\setlength{\BriefingPanelContentWidth}{\BriefingPanelWidth - 2\BriefingPanelPadding}
	% ------------------------------------------------------------------------
	% Panel
	% ------------------------------------------------------------------------
	\par\vspace{6pt plus 2pt minus 2pt}\noindent
	\begin{tikzpicture}
		% --------------------------------------------------------------------
		% Content
		% --------------------------------------------------------------------
		\node[
			anchor=north~west,
			inner~xsep=\BriefingPanelPadding,
			inner~ysep=\BriefingPanelPadding,
			outer~sep=0pt
		] (frame) at (0,0)
		{%
			\begin{minipage}{\BriefingPanelContentWidth}
				\group_begin:
				% ------------------------------------------------------------
				% Header
				% ------------------------------------------------------------
				\centering\vspace{-3pt}
				\ifthenelse{\equal{\BriefingPanelCrest}{}}{}{%
					\BriefingPanelCrest\par\vspace{3pt}
				}%
				
				\ifthenelse{\equal{\BriefingPanelTitle}{}}{}{%
					\color{briefingpanelbluedark}\Large\bfseries\scshape\DndContour{\BriefingPanelTitle}\par
				}%
				
				\ifthenelse{\equal{\BriefingPanelSubTitle}{}}{}{%
					\vspace{1pt}\color{briefingpanelblue}\small\scshape\BriefingPanelSubTitle\par
				}%
				
				\ifthenelse{\equal{\BriefingPanelDivider}{}}{}{%
					\vspace{1pt}\BriefingPanelDivider\vspace{-5pt}
				}%
				\group_end:
				% ------------------------------------------------------------
				% Entries
				% ------------------------------------------------------------
				#2
			\end{minipage}
		};%
		% --------------------------------------------------------------------
		% Chamfered Double Outline
		% --------------------------------------------------------------------		
		% --- outer frame coordinates
		\coordinate (briefingnorthwest)
			at ([shift={(0.8pt,-0.8pt)}]frame.north~west);
		\coordinate (briefingnortheast)
			at ([shift={(-0.8pt,-0.8pt)}]frame.north~east);
		\coordinate (briefingsoutheast)
			at ([shift={(-0.8pt,0.8pt)}]frame.south~east);
		\coordinate (briefingsouthwest)
			at ([shift={(0.8pt,0.8pt)}]frame.south~west);
		
		% --- inner frame coordinates
		\coordinate (innernorthwest)
			at ([shift={(\BriefingPanelInnerInset,-\BriefingPanelInnerInset)}]briefingnorthwest);
		\coordinate (innernortheast)
			at ([shift={(-\BriefingPanelInnerInset,-\BriefingPanelInnerInset)}]briefingnortheast);
		\coordinate (innersoutheast)
			at ([shift={(-\BriefingPanelInnerInset,\BriefingPanelInnerInset)}]briefingsoutheast);
		\coordinate (innersouthwest)
			at ([shift={(\BriefingPanelInnerInset,\BriefingPanelInnerInset)}]briefingsouthwest);
		
		% --- outer frame
		\draw[
			color=briefingpanelbluedark,
			line~width=3pt,
			line~join=miter
		]
			([xshift=\BriefingPanelChomfer]briefingnorthwest)
			-- ([xshift=-\BriefingPanelChomfer]briefingnortheast)
			-- ([yshift=-\BriefingPanelChomfer]briefingnortheast)
			-- ([yshift=\BriefingPanelChomfer]briefingsoutheast)
			-- ([xshift=-\BriefingPanelChomfer]briefingsoutheast)
			-- ([xshift=\BriefingPanelChomfer]briefingsouthwest)
			-- ([yshift=\BriefingPanelChomfer]briefingsouthwest)
			-- ([yshift=-\BriefingPanelChomfer]briefingnorthwest)
			-- cycle;
		
		% --- inner frame
		\draw[
			color=briefingpanelblue,
			line~width=0.5pt,
			line~join=miter,
			opacity=0.85
		]
			([xshift=\BriefingPanelInnerChomfer]innernorthwest)
			-- ([xshift=-\BriefingPanelInnerChomfer]innernortheast)
			-- ([yshift=-\BriefingPanelInnerChomfer]innernortheast)
			-- ([yshift=\BriefingPanelInnerChomfer]innersoutheast)
			-- ([xshift=-\BriefingPanelInnerChomfer]innersoutheast)
			-- ([xshift=\BriefingPanelInnerChomfer]innersouthwest)
			-- ([yshift=\BriefingPanelInnerChomfer]innersouthwest)
			-- ([yshift=-\BriefingPanelInnerChomfer]innernorthwest)
			-- cycle;
	\end{tikzpicture}
	\par
	\vspace{6pt plus 2pt minus 2pt}
}{\group_end:}%

				-- (\a-8:\ToothOuterRadius)
				-- (\a+8:\ToothOuterRadius)
				-- (\a+8:\OuterRingRadius);
		}

		% ---------------------------------------------------------------------
		% Gear ring
		% ---------------------------------------------------------------------
		\draw[
			color=briefingpanelbluedark,
			line width=0.65pt
		]
			(0,0) circle[radius=\OuterRingRadius];

		\draw[
			color=briefingpanelblue,
			line width=0.65pt
		]
			(0,0) circle[radius=\InnerRingRadius];

		% ---------------------------------------------------------------------
		% Minimal compass needle
		% ---------------------------------------------------------------------
		\draw[
			color=briefingpanelbluelight,
			line width=0.6pt
		]
			(0,\NeedleLong)
			-- (\NeedleShort,0)
			-- (0,-\NeedleLong)
			-- (-\NeedleShort,0)
			-- cycle;

		% small center
		\fill[briefingpanelblue]
			(0,0) circle[radius=0.8pt];
