\newcommand{\EncounterPanelCorner}[3]{%
	\pgfmathsetlengthmacro{\EncounterCornerXShift}{%
		#2 * 0.175 * \EncounterPanelCornerSize%
	}%
	\pgfmathsetlengthmacro{\EncounterCornerYShift}{%
		#3 * -1 * 0.175 * \EncounterPanelCornerSize%
	}%
	
	\node[transform shape, anchor=#1]
		at ($ (frame.#1) + (\EncounterCornerXShift,\EncounterCornerYShift) $) {%
		\scalebox{#2}[#3]{%
			\includegraphics[width=\EncounterPanelCornerSize, keepaspectratio]{%
				\PATH templates/utilities/images/backgrounds/encounter/corner-ornament.png
			}%
		}%
	};%
}%

	\EncounterPanelCorner{south west}{-1}{1}
	% Bottom right corner
	\EncounterPanelCorner{south east}{1}{1}
}%

	% Banner
	% ------------------------------------------------------------------------
	\node[inner sep=0pt] (EncounterBanner) at ($ (frame.north) + (0,\EncounterBannerLift) $) {%
		\includegraphics[width=\EncounterBannerWidth]{%
			\PATH templates/utilities/images/backgrounds/encounter/encounter-banner.png%
		}%
	};%
	% ------------------------------------------------------------------------
	% Title
	% ------------------------------------------------------------------------
	\def\BannerTitle{#1}
	\path[
        decorate,
        decoration={
            text along path,
            text align={center},
            raise=-0.2ex,
            text format delimiters={|}{|},
            text={|\normalsize\bfseries\scshape\color{white}|\BannerTitle}
        }
    ]
    ($(EncounterBanner.west)+(8mm,-2mm)$)
        .. controls
            ($(EncounterBanner.center)+(0,2mm)$)
            and
            ($(EncounterBanner.center)+(0,2mm)$)
        ..
    ($(EncounterBanner.east)+(-8mm,-2mm)$);
    
    \node at ($(EncounterBanner.center)+(0,-6mm)$) {\tiny\scshape peril shapes story};
}

\newcommand{\EncounterBlockFramePath}{%
	% Top edge
	($(frame.north west)+(\EncounterBlockCornerSize,0)$)
	--
	($(frame.north east)+(-\EncounterBlockCornerSize,0)$)

	% Top-right inward corner
	.. controls
		($(frame.north east)+(-\EncounterBlockCornerSize,-\EncounterBlockCornerSize)$)
		and
		($(frame.north east)+(0,-\EncounterBlockCornerSize)$)
	..
	($(frame.north east)+(0,-\EncounterBlockCornerSize)$)

	% Right edge
	--
	($(frame.south east)+(0,\EncounterBlockCornerSize)$)

	% Bottom-right inward corner
	.. controls
		($(frame.south east)+(-\EncounterBlockCornerSize,\EncounterBlockCornerSize)$)
		and
		($(frame.south east)+(-\EncounterBlockCornerSize,0)$)
	..
	($(frame.south east)+(-\EncounterBlockCornerSize,0)$)

	% Bottom edge
	--
	($(frame.south west)+(\EncounterBlockCornerSize,0)$)

	% Bottom-left inward corner
	.. controls
		($(frame.south west)+(\EncounterBlockCornerSize,0)$)
		and
		($(frame.south west)+(\EncounterBlockCornerSize,\EncounterBlockCornerSize)$)
	..
	($(frame.south west)+(0,\EncounterBlockCornerSize)$)

	% Left edge
	--
	($(frame.north west)+(0,-\EncounterBlockCornerSize)$)

	% Top-left inward corner
	.. controls
		($(frame.north west)+(0,-\EncounterBlockCornerSize)$)
		and
		($(frame.north west)+(\EncounterBlockCornerSize,-\EncounterBlockCornerSize)$)
	..
	($(frame.north west)+(\EncounterBlockCornerSize,0)$)

	-- cycle
}

\NewTColorBox{EncounterBlock}{ O{} m }{%
	enhanced,
	width=\linewidth,
	% ------------------------------------------------------------------------
	colback=white,
	opacityback=0,
	% ------------------------------------------------------------------------
	% Disable ordinary tcolorbox frame
	% ------------------------------------------------------------------------
	frame hidden,
	boxrule=0pt,
	sharp corners,
	% ------------------------------------------------------------------------
	% Minimum height
	% ------------------------------------------------------------------------
	height from={\EncounterBlockMinHeight} to {\textheight},
	% ------------------------------------------------------------------------
	boxsep=0pt,
	left={\dimexpr \logowidthmul\linewidth + \EncounterBlockCornerSize + 10pt\relax},
	right=40pt,
	top={\dimexpr 2.5\EncounterBlockCornerSize + 6pt \relax},
	bottom=7pt,
	% ------------------------------------------------------------------------
	before skip=4pt,
	after skip=4pt,
	fontupper=\scriptsize,
	% ------------------------------------------------------------------------
	% Reset Difficulty Value
	% ------------------------------------------------------------------------
	before upper={%
		\gdef\EncounterDifficultyValue{}%
	},
	% ------------------------------------------------------------------------
	% Underlying background, header banner and bezel outline
	% ------------------------------------------------------------------------
	underlay={%
		% --------------------------------------------------------------------
		% Background
		% --------------------------------------------------------------------
		\fill[encounterblockbackground] \EncounterBlockFramePath;
		\begin{scope}
			\path[clip] \EncounterBlockFramePath;
			\node[anchor=south east, opacity=0.45, inner sep=0pt] at (frame.south east) {\EncounterBlockBackground{#2}};
		\end{scope}
		% --------------------------------------------------------------------
		% Header Banner
		% --------------------------------------------------------------------
		\filldraw[
			fill=encounterpanelred,
			draw=encounterblockdark,
			line width=\EncounterBlockFrameWidth,
			line join=round
		]
			\EncounterHeaderBlockFramePath;
		\draw[
			draw=encounterblockgold,
			line width=0.55,
			line join=round
		]
			\EncounterHeaderBlockFramePath;
		% --------------------------------------------------------------------
		% Dark heavy bezel
		% --------------------------------------------------------------------
		\draw[
			draw=encounterblockdark,
			line width=\EncounterBlockFrameWidth,
			line join=round
		]
			\EncounterBlockFramePath;
		% --------------------------------------------------------------------
		% Logo
		% --------------------------------------------------------------------
		\filldraw[
			fill=encounterpanelred,
			draw=encounterblockdark,
			line width=\EncounterBlockFrameWidth,
			line join=round
		]
			\EncounterLogoBlockFramePath;
	},
	% ------------------------------------------------------------------------
	% Finish
	% ------------------------------------------------------------------------
	finish={%
		% --------------------------------------------------------------------
		% Header
		% --------------------------------------------------------------------
		\node[
    		anchor=west,
    		inner sep=0pt,
    		text=white,
    		font=\large\bfseries\scshape
		] at (
			$(frame.north west)
				+(\logowidthmul\linewidth,0)
				+(2.25\EncounterBlockCornerSize,-1.25\EncounterBlockCornerSize) $
		) {%
			\EncounterBlockTitle{#2}%
		};
		% --------------------------------------------------------------------
		% Logo
		% --------------------------------------------------------------------
        \coordinate (EncounterLogoRight) at (
			$ (frame.west)
				+(\logowidthmul\linewidth,0)
				+(\EncounterBlockCornerSize,0) $
		);
		\node[inner sep=0pt]
			at ($(frame.west)!0.5!(EncounterLogoRight)$) {%
				\EncounterBlockLogo{#2}%
		};
		% --------------------------------------------------------------------
		% Thin gold highlight through the bezel
		% --------------------------------------------------------------------
		\draw[
			draw=encounterblockgold,
			line width=0.65pt,
			line join=round
		]
			\EncounterBlockFramePath;
		% --------------------------------------------------------------------
		% Logo
		% --------------------------------------------------------------------
		\draw[
			draw=encounterblockgold,
			line width=0.65pt,
			line join=round
		]
			\EncounterLogoBlockFramePath;
		% --------------------------------------------------------------------
		% Difficulty indicator, if applicable
		% --------------------------------------------------------------------
    	\EncounterDrawDifficulty{#2}
	},
	% ------------------------------------------------------------------------
	#1%
}%

\NewDocumentCommand{\EncounterDrawDifficulty}{m}
{%
    \str_if_eq:nnT {#1} {difficulty}%
    {%
        \tl_if_blank:eF { \EncounterDifficultyValue }%
        {%
            \EncounterDifficultyPipNode%
            {%
                \EncounterDifficultyDisplay%
                {%
                    \EncounterDifficultyValue%
                }%
            }%
        }%
    }%
}%
\NewDocumentCommand{\EncounterDifficultyDisplay}{m}
{
	\str_case_e:nnF {#1}
	{
		{Trivial}  { \EncounterDifficultyPips{1} }
		{Easy}     { \EncounterDifficultyPips{2} }
		{Moderate} { \EncounterDifficultyPips{3} }
		{Hard}     { \EncounterDifficultyPips{4} }
		{Deadly}   { \EncounterDifficultyPips{5} }
	}
	{
		\PackageError
